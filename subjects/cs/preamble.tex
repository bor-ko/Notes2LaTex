% Statement boxes with a title bar. Optional name: \begin{definition}[Big-O]
% #1 environment, #2 label, #3 colour
\newcommand{\csbox}[3]{%
  \newtcolorbox{#1}[1][]{breakable, colback=#3!4, colframe=#3!50!black,
    fonttitle=\bfseries, boxrule=0.6pt, arc=2pt,
    title={#2\if\relax\detokenize{##1}\relax\else\ (##1)\fi}}}
\csbox{definition}{Definition}{blue}
\csbox{theorem}{Theorem}{red}
\csbox{lemma}{Lemma}{red}
\csbox{proposition}{Proposition}{red}
\csbox{corollary}{Corollary}{red}
\csbox{example}{Example}{green}
\csbox{remark}{Remark}{gray}
\csbox{note}{Note}{orange}
\newenvironment{proof}{\par\noindent\textit{Proof.}\ }{\hfill$\square$\par}
